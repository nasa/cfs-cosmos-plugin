<% require 'cfs_packet_headers' %>
<% require 'cfs_globals' %>

# To set the memory size, uncomment the lines below

# <% cfs_globals_set_mem_size(cfs_mem_size) %>

# To expose the endianness config to the plugin install, uncomment the lines below

# <% cfs_globals_set_endianness(cfs_endianness) %>

# This should map to the IP address that the FSW is running from
VARIABLE cfs_intf_ip X.X.X.X

# This should map to the CPU number of the FSW to connect to
VARIABLE cfs_cpu_idx 1

# Declare a new instance of the CFS target
TARGET CFS <%= cfs_target_name_from_cpu_num(cfs_cpu_idx) %>

# Set up the FSW cmd/tlm interface
INTERFACE DTNFSW_<%= cfs_cpu_idx %>_INT udp_interface.rb &
        <%= cfs_intf_ip %> &
        <%= cfs_get_port_num_cmd(cfs_cpu_idx) %> &
        <%= cfs_get_port_num_tlm() %> &
        nil nil 128 nil nil
    MAP_TARGET <%= cfs_target_name_from_cpu_num(cfs_cpu_idx) %>
    PROTOCOL WRITE ProtocolCcsdsChecksum
    PROTOCOL WRITE ProtocolCcsdsLength
    PROTOCOL READ LengthProtocol 32 16 7 1 BIG_ENDIAN 0 nil
    PROTOCOL READ ProtocolEvsLog
